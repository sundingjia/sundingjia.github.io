```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5+10;
struct node1
{
    int h,w;
    int id;
    bool friend operator<(node1 a,node1 b)
    {
        return a.h<b.h;
    }
};
struct node2
{
    int h,w;
    int id;
    bool friend operator<(node2 a,node2 b)
    {
        return a.w<b.w;
    }
};
priority_queue<node1> q1;
priority_queue<node2> q2;
int n;
int x[N],y[N];
int del[N];
int posx[N],posy[N];
void solve(int h,int w,int x,int y)
{
    // cout<<h<<' '<<w<<' '<<x<<' '<<y<<'\n';
    while(q1.size()&&del[q1.top().id])
    {
        q1.pop();
    }
    while(q2.size()&&del[q2.top().id])
    {
        q2.pop();
    }
    if(q1.size())
    {
        if(q1.top().h==h)
        {
            del[q1.top().id]=1;
            posx[q1.top().id]=x;
            posy[q1.top().id]=y;
            int nw=w-q1.top().w;
            int ny=y+q1.top().w;
            q1.pop();
            solve(h,nw,x,ny);
            return;
        }
    }
    if(q2.size())
    {
        if(q2.top().w==w)
        {
            del[q2.top().id]=1;
            posx[q2.top().id]=x;
            posy[q2.top().id]=y;
            int nh=h-q2.top().h;
            int nx=x+q2.top().h;
            q2.pop();
            solve(nh,w,nx,y);
            return;            
        }
    }
    return;
}
int main()
{
    int h,w;
    cin>>h>>w>>n;
    for(int i=1;i<=n;i++)
    {
        cin>>x[i]>>y[i];
        q1.push(node1{x[i],y[i],i});
        q2.push(node2{x[i],y[i],i});
    }
    solve(h,w,1,1);
    for(int i=1;i<=n;i++)
    {
        cout<<posx[i]<<' '<<posy[i]<<'\n';
    }
    return 0;
}
```

```cpp
    long long res=1;

    {
        if(y&1) 
        {
            res*=x;res%=Mod;
        }
        x*=x;x%=Mod;y/=2;
    }
    return res;
}
```
